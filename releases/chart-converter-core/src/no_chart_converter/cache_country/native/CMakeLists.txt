cmake_minimum_required(VERSION 3.22)
project(no5_s52_provider LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(NO5_MAP_SOURCE "" CACHE PATH "Explicit external no-apps/map source directory")
if(NOT EXISTS "${NO5_MAP_SOURCE}/chart-engine/CMakeLists.txt")
  message(FATAL_ERROR "Set NO5_MAP_SOURCE to the independently checked public No5 Map source tree")
endif()
set(NO5_BUILD_TESTS OFF CACHE BOOL "No host test matrix for provider build" FORCE)
set(NO5_VIEWPORT_REUSE OFF CACHE BOOL "Provider resolves rules, without GPU rendering" FORCE)
add_subdirectory("${NO5_MAP_SOURCE}/chart-engine" no5-engine)
add_executable(no5_s52_provider s52_provider.cpp)
target_link_libraries(no5_s52_provider PRIVATE no5_chart_core ${NO5_HOST_GL} ${NO5_HOST_EGL})
